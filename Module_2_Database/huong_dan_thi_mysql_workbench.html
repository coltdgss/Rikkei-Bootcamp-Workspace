CREATE DATABASE IF NOT EXISTS CuaHangKobani;
USE CuaHangKobani;

-- Tắt safe updates để thực thi UPDATE/DELETE mượt mà
SET SQL_SAFE_UPDATES = 0;

-- 1.1. Bảng KhachHang (Khách hàng)
CREATE TABLE IF NOT EXISTS KhachHang (
    khachhang_id VARCHAR(10) PRIMARY KEY,
    ho_ten VARCHAR(100) NOT NULL,
    email VARCHAR(100) NOT NULL UNIQUE,
    so_dien_thoai VARCHAR(15) NOT NULL,
    dia_chi VARCHAR(200)
);

-- 1.2. Bảng SanPham (Sản phẩm kinh doanh)
CREATE TABLE IF NOT EXISTS SanPham (
    sanpham_id VARCHAR(10) PRIMARY KEY,
    ten_san_pham VARCHAR(150) NOT NULL UNIQUE,
    gia_ban DECIMAL(12,2) NOT NULL CHECK (gia_ban > 0),
    trang_thai VARCHAR(50) NOT NULL DEFAULT 'Đang bán',
    so_luong_ton INT NOT NULL
);

-- 1.3. Bảng DonHang (Đơn đặt hàng - tham chiếu KhachHang và SanPham)
CREATE TABLE IF NOT EXISTS DonHang (
    donhang_id INT PRIMARY KEY AUTO_INCREMENT,
    khachhang_id VARCHAR(10) NOT NULL,
    sanpham_id VARCHAR(10) NOT NULL,
    so_luong INT NOT NULL CHECK (so_luong > 0),
    ngay_dat DATE NOT NULL,
    ngay_giao DATE,
    tong_tien DECIMAL(12,2) DEFAULT 0,
    CONSTRAINT fk_donhang_khachhang FOREIGN KEY (khachhang_id) REFERENCES KhachHang(khachhang_id),
    CONSTRAINT fk_donhang_sanpham FOREIGN KEY (sanpham_id) REFERENCES SanPham(sanpham_id)
);

-- 1.4. Bảng ThanhToan (Lịch sử thanh toán - tham chiếu DonHang)
CREATE TABLE IF NOT EXISTS ThanhToan (
    thanhtoan_id INT PRIMARY KEY AUTO_INCREMENT,
    donhang_id INT NOT NULL,
    phuong_thuc_tt VARCHAR(50) NOT NULL,
    ngay_tt DATE NOT NULL,
    so_tien_tt DECIMAL(12,2) NOT NULL CHECK (so_tien_tt > 0),
    CONSTRAINT fk_thanhtoan_donhang FOREIGN KEY (donhang_id) REFERENCES DonHang(donhang_id)
);

-- 1.6. CẬP NHẬT DỮ LIỆU (UPDATE CÓ JOIN)

UPDATE DonHang dh
JOIN SanPham sp ON dh.sanpham_id = sp.sanpham_id
SET dh.tong_tien = (sp.gia_ban * dh.so_luong) + 20.00
WHERE sp.trang_thai = 'Đang bán'
  AND dh.ngay_dat < CURDATE();

-- =============================================================================
-- 1.7. XÓA DỮ LIỆU (DELETE)
-- =============================================================================
DELETE FROM ThanhToan
WHERE phuong_thuc_tt = 'Cash'
  AND so_tien_tt < 150.00;

-- Câu 1: Khách hàng sắp xếp theo họ tên tăng dần
SELECT khachhang_id, ho_ten, email, so_dien_thoai, dia_chi
FROM KhachHang ORDER BY ho_ten ASC;

-- Câu 2: Sản phẩm sắp xếp theo giá bán giảm dần
SELECT sanpham_id, ten_san_pham, gia_ban, so_luong_ton
FROM SanPham ORDER BY gia_ban DESC;

-- Câu 3: Khách hàng và sản phẩm đã đặt (JOIN 3 bảng)
SELECT kh.khachhang_id, kh.ho_ten, sp.sanpham_id, dh.ngay_dat, dh.ngay_giao
FROM DonHang dh
INNER JOIN KhachHang kh ON dh.khachhang_id = kh.khachhang_id
INNER JOIN SanPham sp ON dh.sanpham_id = sp.sanpham_id;

-- Câu 4: Khách hàng và số tiền đã thanh toán, giảm dần
SELECT kh.khachhang_id, kh.ho_ten, tt.phuong_thuc_tt, tt.so_tien_tt
FROM ThanhToan tt
INNER JOIN DonHang dh ON tt.donhang_id = dh.donhang_id
INNER JOIN KhachHang kh ON dh.khachhang_id = kh.khachhang_id
ORDER BY tt.so_tien_tt DESC;

-- Câu 5: Lấy thông tin khách hàng từ vị trí 2 đến 4
SELECT * FROM KhachHang ORDER BY ho_ten ASC LIMIT 3 OFFSET 1;

-- Câu 6: Khách hàng đặt ít nhất 2 đơn VÀ tổng thanh toán > 500
SELECT kh.khachhang_id, kh.ho_ten, COUNT(DISTINCT dh.donhang_id) AS so_luong_don_hang
FROM KhachHang kh
INNER JOIN DonHang dh ON kh.khachhang_id = dh.khachhang_id
INNER JOIN ThanhToan tt ON dh.donhang_id = tt.donhang_id
GROUP BY kh.khachhang_id, kh.ho_ten
HAVING COUNT(DISTINCT dh.donhang_id) >= 2 AND SUM(tt.so_tien_tt) > 500.00;

-- Câu 7: Sản phẩm tổng tiền < 1000 và ít nhất 2 khách hàng khác nhau
SELECT sp.sanpham_id, sp.ten_san_pham, sp.gia_ban, SUM(tt.so_tien_tt) AS tong_tien_thanh_toan
FROM SanPham sp
INNER JOIN DonHang dh ON sp.sanpham_id = dh.sanpham_id
INNER JOIN ThanhToan tt ON dh.donhang_id = tt.donhang_id
GROUP BY sp.sanpham_id, sp.ten_san_pham, sp.gia_ban
HAVING SUM(tt.so_tien_tt) < 1000.00 AND COUNT(DISTINCT dh.khachhang_id) >= 2;

-- Câu 8: Khách hàng có tổng số tiền thanh toán > 500
SELECT kh.khachhang_id, kh.ho_ten, SUM(tt.so_tien_tt) AS tong_tien_thanh_toan
FROM KhachHang kh
INNER JOIN DonHang dh ON kh.khachhang_id = dh.khachhang_id
INNER JOIN ThanhToan tt ON dh.donhang_id = tt.donhang_id
GROUP BY kh.khachhang_id, kh.ho_ten
HAVING SUM(tt.so_tien_tt) > 500.00;

-- Câu 9: Khách hàng có họ tên chứa 'Minh' HOẶC địa chỉ ở 'Hanoi'
SELECT khachhang_id, ho_ten, email, so_dien_thoai, dia_chi
FROM KhachHang
WHERE ho_ten LIKE '%Minh%' OR dia_chi LIKE '%Hanoi%'
ORDER BY ho_ten ASC;

-- Câu 10: Sản phẩm giá bán giảm dần, lấy 3 sản phẩm tiếp theo sau 2 sản phẩm đầu (Trang 2)
SELECT sanpham_id, ten_san_pham, gia_ban
FROM SanPham ORDER BY gia_ban DESC LIMIT 3 OFFSET 2;

CREATE OR REPLACE VIEW view_sanpham_khachhang_dat_hang AS
SELECT sp.sanpham_id, sp.ten_san_pham, kh.khachhang_id, kh.ho_ten
FROM DonHang dh
INNER JOIN SanPham sp ON dh.sanpham_id = sp.sanpham_id
INNER JOIN KhachHang kh ON dh.khachhang_id = kh.khachhang_id
WHERE dh.ngay_dat < '2026-09-06';

CREATE OR REPLACE VIEW view_khachhang_donhang_gia_cao AS

FROM DonHang dh
INNER JOIN KhachHang kh ON dh.khachhang_id = kh.khachhang_id
INNER JOIN SanPham sp ON dh.sanpham_id = sp.sanpham_id

-- =============================================================================
-- PHẦN 4: TẠO TRIGGER (BẪY LỖI & TỰ ĐỘNG HÓA SỰ KIỆN)
-- =============================================================================

-- Trigger 1: Báo lỗi nếu ngay_giao < ngay_dat
DROP TRIGGER IF EXISTS check_insert_donhang;
DELIMITER $$
CREATE TRIGGER check_insert_donhang
BEFORE INSERT ON DonHang
FOR EACH ROW
BEGIN
    IF NEW.ngay_giao IS NOT NULL AND NEW.ngay_giao < NEW.ngay_dat THEN
        SIGNAL SQLSTATE '45000'
        SET MESSAGE_TEXT = 'Ngày giao không thể trước ngày đặt hàng được !';
    END IF;
END$$
DELIMITER ;

-- Trigger 2: Tự động đổi 'Ngừng bán' khi sản phẩm được đặt quá 30 lần trong ngày
DROP TRIGGER IF EXISTS update_sanpham_status_on_order;
DELIMITER $$
CREATE TRIGGER update_sanpham_status_on_order
AFTER INSERT ON DonHang
FOR EACH ROW
BEGIN
    DECLARE total_orders_today INT;
    SELECT COUNT(*) INTO total_orders_today
    FROM DonHang
    WHERE sanpham_id = NEW.sanpham_id AND ngay_dat = NEW.ngay_dat;
    
    IF total_orders_today > 30 THEN
        UPDATE SanPham SET trang_thai = 'Ngừng bán' WHERE sanpham_id = NEW.sanpham_id;
    END IF;
END$$
DELIMITER ;

-- =============================================================================
-- PHẦN 5: STORED PROCEDURE (THỦ TỤC LƯU TRỮ)
-- =============================================================================

DROP PROCEDURE IF EXISTS add_khachhang;
DELIMITER $$
CREATE PROCEDURE add_khachhang (
    IN p_khachhang_id VARCHAR(10),
    IN p_ho_ten VARCHAR(100),
    IN p_email VARCHAR(100),
    IN p_so_dien_thoai VARCHAR(15),
    IN p_dia_chi VARCHAR(200)
)
BEGIN
    INSERT INTO KhachHang (khachhang_id, ho_ten, email, so_dien_thoai, dia_chi)
    VALUES (p_khachhang_id, p_ho_ten, p_email, p_so_dien_thoai, p_dia_chi);
END$$
DELIMITER ;

DROP PROCEDURE IF EXISTS add_thanhtoan;
DELIMITER $$
CREATE PROCEDURE add_thanhtoan (
    IN p_donhang_id INT,
    IN p_phuong_thuc_tt VARCHAR(50),
    IN p_so_tien_tt DECIMAL(12,2),
    IN p_ngay_tt DATE
)
BEGIN
    INSERT INTO ThanhToan (donhang_id, phuong_thuc_tt, so_tien_tt, ngay_tt)
    VALUES (p_donhang_id, p_phuong_thuc_tt, p_so_tien_tt, p_ngay_tt);
END$$
DELIMITER ;

Yêu cầu Phần 1
  1. Tạo bảng: Tạo 4 bảng với cấu trúc, khóa chính, khóa ngoại, ràng buộc như mô tả.
  2. Chèn dữ liệu: Thêm dữ liệu vào 4 bảng (sử dụng 8 KH, 8 SP, 10 Đơn hàng, 10 Thanh toán).
  3. Cập nhật dữ liệu: Viết UPDATE tính tong_tien = gia_ban * so_luong + 20.0 cho các đơn có sản phẩm ở trạng thái 'Đang bán' và ngay_dat < CURDATE() (dùng JOIN).
  4. Xóa dữ liệu: Viết DELETE xóa thanh toán nếu phuong_thuc_tt = 'Cash' và so_tien_tt < 150.0.
PHẦN 2: Truy vấn dữ liệu (10 câu)
  1. Lấy thông tin khách hàng, sắp xếp họ tên tăng dần.
  2. Lấy thông tin sản phẩm, sắp xếp giá bán giảm dần.
  3. Lấy thông tin khách hàng và sản phẩm đã đặt (mã KH, họ tên, mã SP, ngày đặt, ngày giao).
  4. Lấy danh sách khách hàng và số tiền đã thanh toán, sắp xếp số tiền thanh toán giảm dần.
  5. Lấy thông tin khách hàng từ vị trí thứ 2 đến thứ 4 theo họ tên.
  6. Lấy khách hàng đã đặt ít nhất 2 đơn và tổng thanh toán trên 500.0.
  7. Lấy sản phẩm có tổng tiền thanh toán dưới 1000.0 và có ít nhất 2 khách hàng khác nhau.
  8. Lấy khách hàng có tổng tiền thanh toán lớn hơn 500.0.
  9. Lấy khách hàng có họ tên chứa "Minh" hoặc địa chỉ ở "Hanoi" (sắp xếp họ tên tăng dần).
  10. Lấy danh sách sản phẩm, sắp xếp giá bán giảm dần. Hiển thị 3 sản phẩm tiếp theo sau 2 sản phẩm đầu (trang thứ 2).
PHẦN 3: Tạo View
  1. Tạo view lấy sản phẩm và khách hàng đã đặt hàng trước ngày 2026-09-06.
  2. Tạo view lấy khách hàng và đơn hàng đã đặt với giá bán lớn hơn 200.0.
PHẦN 4: Tạo Trigger
  1. Tạo trigger check_insert_donhang kiểm tra nếu ngày giao trước ngày đặt thì báo lỗi: "Ngày giao không thể trước ngày đặt hàng được !" và hủy thao tác.
  2. Tạo trigger update_sanpham_status_on_order tự động cập nhật sản phẩm thành 'Ngừng bán' khi đặt quá 30 lần trong cùng 1 ngày.
PHẦN 5: Tạo Stored Procedure
  1. Viết stored procedure add_khachhang thêm mới một khách hàng.
  2. Viết stored procedure add_thanhtoan thêm một thanh toán mới cho một đơn hàng (tham số: p_donhang_id, p_phuong_thuc_tt, p_so_tien_tt, p_ngay_tt).

3 cách mở khung soạn thảo lệnh nhanh chóng và chuyên nghiệp

Để bắt đầu viết các câu lệnh SQL giải bài thi, bạn cần mở một trang soạn thảo trắng (Query Tab). Hãy sử dụng 1 trong 3 cách sau:

⭐ Cách 1 (Nhanh nhất - Chuẩn chuyên gia): Phím tắt

Nhấn tổ hợp phím: Ctrl + T

Ngay lập tức một tab mới xuất hiện bên cạnh tab cũ (ví dụ SQL File 1 hoặc SQL File 4* như trong ảnh thực tế của bạn).

2 Cách 2: Nút bấm trên thanh công cụ (Toolbar)

Nhìn lên thanh công cụ nằm ngay bên dưới menu File / Edit. Bạn sẽ thấy biểu tượng hình tờ giấy có hình tia sét và dấu cộng nhỏ ở góc (tooltip: "Create a new SQL tab for executing queries").

🖱️ Thao tác: Click chuột trái 1 lần vào biểu tượng SQL+ đó.
3 Cách 3: Menu File trên cùng

Trên thanh menu trên cùng của cửa sổ MySQL Workbench:
• Click vào menu File ở góc trên bên trái.
• Chọn dòng đầu tiên: New Query Tab.

🖱️ Thao tác: File ➔ New Query Tab

Chặng 3: Dọn dẹp sạch sẽ bộ dữ liệu cũ (Clean Slate & Reset)
Xóa các database bài tập cũ để tránh lỗi trùng bảng và điểm thi bị sai lệch
Tại sao phải xóa dữ liệu cũ? Trong ảnh thực tế của bạn, máy đang có sẵn các database từ các buổi học trước như: nippon_coffee, rippon_coffee, student_db. Nếu bạn làm bài thi mà chưa dọn dẹp hoặc chạy đè lên database cũ, MySQL sẽ báo lỗi: "Table 'KhachHang' already exists" hoặc truy vấn ra nhầm dữ liệu của bài cũ khiến bài thi bị chấm điểm sai!
⚡ Phương pháp 1: Dọn dẹp bằng Code SQL (Khuyên dùng trong phòng thi)

Gõ các câu lệnh này ngay trên dòng đầu tiên của tab soạn thảo rồi bấm Ctrl + Enter:

-- 1. Xóa bài thi cũ nếu đã tạo dở trước đó
DROP DATABASE IF EXISTS CuaHangKobani;

-- 2. Xóa các database rác buổi học trước (nếu muốn)
DROP DATABASE IF EXISTS student_db;
DROP DATABASE IF EXISTS nippon_coffee;
DROP DATABASE IF EXISTS rippon_coffee;

-- 3. Tạo CSDL bài thi mới trắng tinh và USE ngay
CREATE DATABASE CuaHangKobani;
USE CuaHangKobani;

👉 Nhờ có mệnh đề IF EXISTS, dù database đó chưa từng tồn tại máy tính cũng không bị báo lỗi đỏ!

🖱️ Phương pháp 2: Xóa bằng chuột trên cây SCHEMAS

Nếu muốn xóa tận gốc các database rác bằng giao diện đồ họa:

  1. Nhìn vào cột trái Navigator, đảm bảo tab dưới đáy đang chọn Schemas.
  2. Nhấp chuột phải vào tên Database muốn xóa (ví dụ student_db hoặc nippon_coffee).
  3. Chọn dòng menu: Drop Schema...
  4. Một hộp thoại cảnh báo hiện lên, bấm nút Drop Now (Xóa vĩnh viễn ngay lập tức).
  5. Bấm nút 🔄 Refresh ở góc trên cột SCHEMAS để cập nhật lại danh sách.
⛔ CẢNH BÁO TỬ HUYỆT: Tuyệt đối KHÔNG ĐƯỢC XÓA 3 database hệ thống: sys, sakila, world! Nếu lỡ tay xóa sys, MySQL sẽ bị hỏng nhân hệ thống và không thể chấm bài được.

1 Lưu file bài làm (.sql)

Nhấn tổ hợp phím Ctrl + S (hoặc vào menu File ➔ Save Script As...).

• Chuột phải chọn Open with ➔ Notepad để xem chắc chắn toàn bộ code SQL đã được ghi vào đĩa trước khi tải lên link nộp bài của Rikkei.

🧠 3. Chiến thuật 5 bước phân tích đề thi (Không bao giờ rớt)

Đề thi Kobani có 4 bảng: KhachHang, SanPham, DonHang, ThanhToan. Muốn không bị lỗi ràng buộc khóa ngoại, hãy làm theo đúng 5 bước sau:

1
Bước 1: Tạo Database và USE ngay dòng đầu tiên
Nếu không gõ lệnh USE CuaHangKobani;, Workbench sẽ báo lỗi đỏ: "Error Code 1046: No database selected".
2
Bước 2: Phân loại bảng CHA vs bảng CON (Thứ tự tạo bảng sống còn)
- Bảng CHA (Không có khóa ngoại FK): Tạo trước! Đó là bảng KhachHang và SanPham.
- Bảng CON (Có khóa ngoại FK tham chiếu): Tạo sau! Đó là DonHang (tham chiếu KhachHang, SanPham) và cuối cùng là ThanhToan (tham chiếu DonHang).
⚠️ Nếu bạn tạo bảng DonHang trước bảng KhachHang, MySQL sẽ báo lỗi vì chưa có bảng cha để tham chiếu!
3
Bước 3: Nhận diện ràng buộc đề bài (Constraints)
- PK: Khóa chính (PRIMARY KEY).
- NN: Không được rỗng (NOT NULL).
- UNIQUE: Không được trùng (UNIQUE như email, ten_san_pham).
- CHECK: Kiểm tra điều kiện (CHECK (gia_ban > 0), CHECK (so_luong > 0)).
- DEFAULT: Giá trị mặc định (DEFAULT 'Đang bán', DEFAULT 0).
- AUTO_INCREMENT: Tự động tăng ID (donhang_id, thanhtoan_id).
4
Bước 4: Thứ tự INSERT chèn dữ liệu mẫu
Tương tự quy tắc tạo bảng: Bảng cha có dữ liệu trước thì bảng con mới có cái để liên kết!
Thứ tự chèn: KhachHang -> SanPham -> DonHang -> ThanhToan.
5
Bước 5: Xử lý UPDATE có JOIN và DELIMITER cho Trigger/Procedure
- Khi câu lệnh UPDATE cần lấy cột từ bảng khác (ví dụ lấy gia_ban từ SanPham để tính tong_tien cho DonHang), bắt buộc dùng cú pháp UPDATE DonHang JOIN SanPham ON ....
- Đối với Trigger và Stored Procedure: Luôn kẹp DELIMITER $$ ở đầu và DELIMITER ; ở cuối.

⚠️ 4. Bẫy điểm liệt thường gặp & Cách gỡ lỗi tức thì

1. Lỗi "Safe Update Mode" (Error 1175 khi chạy câu UPDATE / DELETE):
MySQL Workbench mặc định chặn các câu lệnh UPDATE/DELETE nếu không có khóa chính WHERE id = ....
👉 Cách xử lý: Gõ dòng lệnh này ở đầu file và chạy trước: 
SET SQL_SAFE_UPDATES = 0;
2. Lỗi dấu gạch chân đỏ khi viết Trigger / Procedure:
Do chưa khai báo DELIMITER $$. Khi gặp dấu ; bên trong thân BEGIN...END, MySQL tưởng đã hết lệnh nên báo lỗi syntax.
👉 Cách xử lý: Bắt buộc dùng DELIMITER $$ ở đầu và DELIMITER ; ở cuối khối lệnh.
3. Lỗi "Cannot add foreign key constraint" (Error 1215):
Do bạn tạo bảng con trước bảng cha, hoặc kiểu dữ liệu ở bảng con không giống y hệt bảng cha.
👉 Cách xử lý: Kiểm tra lại: khachhang_id ở cả 2 bảng đều phải là VARCHAR(10).
4. Mẹo kiểm tra bài thi trước khi nộp:
- Chạy lệnh SHOW TABLES; để kiểm tra xem đã đủ 4 bảng chưa.
- Chạy SELECT COUNT(*) FROM KhachHang; để kiểm tra dữ liệu mẫu đã vào đủ 8 dòng chưa.
- Nhấn Ctrl + S để lưu bài và kiểm tra file .sql đã có trong thư mục chưa.
