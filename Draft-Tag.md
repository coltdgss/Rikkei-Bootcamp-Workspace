- // Tạo một biến với giá trị ban đầu là "A"
let name = 'A';
console.log(name); // In ra: A

// Gán lại giá trị cho biến name
name = 'B';
console.log(name); // In ra: B

// Nhưng nếu sử dụng const thì sao?
const PI = 3.14;
console.log(PI); // In ra: 3.14

// Nếu bạn cố gắng gán lại giá trị cho PI, sẽ xảy ra lỗi
PI = 3.14159; // Lỗi syntax
